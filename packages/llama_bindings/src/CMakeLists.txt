cmake_minimum_required(VERSION 3.22)
project(llama_bindings LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ---------------------------------------------------------------------------
# Where llama.cpp comes from
# ---------------------------------------------------------------------------
#
# `scripts/build_llama_android.sh` cross-compiles llama.cpp once and drops the results in
#   packages/llama_bindings/android/prebuilt/<abi>/*.so
#   packages/llama_bindings/android/prebuilt/include/*.h
#
# Building llama.cpp from source on every app build would add 10–20 minutes to a clean CI
# run for no benefit, so this imports the prebuilt libraries instead. Run the script once
# after cloning, and again whenever the pinned llama.cpp commit changes.

set(PREBUILT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../android/prebuilt")
set(PREBUILT_ABI "${PREBUILT_ROOT}/${ANDROID_ABI}")

# `if(EXISTS)` is evaluated once, at configure time, and creates no dependency — so once the
# placeholder branch below has been taken, the generated ninja files keep taking it even
# after scripts/build_llama_android.sh has produced the real .so files, and the app stays on
# StubChatEngine forever. This glob exists solely for its CONFIGURE_DEPENDS side effect: it
# makes the build system re-run CMake configure when the contents of that directory change.
file(GLOB _llama_prebuilt CONFIGURE_DEPENDS "${PREBUILT_ABI}/*.so")

# A missing prebuilt used to be a FATAL_ERROR here, which made a fresh clone unbuildable:
# `flutter run` died during the Gradle configure step, before a single line of Dart ran.
#
# It degrades instead. A placeholder `libllama_shim.so` is built with none of the `lc_*`
# symbols in it, so `DynamicLibrary.open` still succeeds and
# `isLlamaLibraryAvailable()` (packages/llama_bindings/lib/src/llama_library.dart) sees no
# `lc_backend_init` and returns false. `AppCoordinator` then selects `StubChatEngine` and
# the app launches with everything except inference working.
#
# Building a target rather than skipping the library entirely is deliberate: AGP's
# externalNativeBuild expects the CMake project to produce something, and an empty build
# fails in a much less legible way than a library with no symbols.
#
# Run ./scripts/build_llama_android.sh and rebuild to get the real shim back.
if(NOT EXISTS "${PREBUILT_ABI}/libllama.so")
    message(WARNING
        "llama.cpp has not been built for ${ANDROID_ABI} — building a placeholder shim.\n"
        "The app will run with inference disabled (StubChatEngine).\n"
        "To enable it: ./scripts/build_llama_android.sh\n"
        "Expected: ${PREBUILT_ABI}/libllama.so")

    set(PLACEHOLDER_SRC "${CMAKE_CURRENT_BINARY_DIR}/llama_shim_placeholder.cpp")
    file(WRITE "${PLACEHOLDER_SRC}"
        "// Generated by CMakeLists.txt when no prebuilt llama.cpp was found.\n"
        "// Exports no lc_* symbol on purpose: that absence is what the Dart-side probe\n"
        "// detects. Do not add one.\n"
        "extern \"C\" int llama_shim_placeholder(void) { return 0; }\n")

    add_library(llama_shim SHARED "${PLACEHOLDER_SRC}")
    return()
endif()

# llama.cpp splits into several shared objects. Every one of them has to be imported and
# installed into the APK, or the loader fails at runtime with a missing-dependency error
# that names only the top-level library.
set(LLAMA_LIBS llama ggml ggml-base ggml-cpu)

foreach(LIB ${LLAMA_LIBS})
    add_library(${LIB} SHARED IMPORTED)
    set_target_properties(${LIB} PROPERTIES
        IMPORTED_LOCATION "${PREBUILT_ABI}/lib${LIB}.so")
endforeach()

# Optional Adreno OpenCL backend. Present only when the build script was run with
# LLAMA_ANDROID_OPENCL=1, and Snapdragon-only even then.
if(EXISTS "${PREBUILT_ABI}/libggml-opencl.so")
    add_library(ggml-opencl SHARED IMPORTED)
    set_target_properties(ggml-opencl PROPERTIES
        IMPORTED_LOCATION "${PREBUILT_ABI}/libggml-opencl.so")
    list(APPEND LLAMA_LIBS ggml-opencl)
    message(STATUS "llama_bindings: OpenCL backend included")
endif()

# ---------------------------------------------------------------------------
# The shim
# ---------------------------------------------------------------------------

add_library(llama_shim SHARED llama_shim.cpp)

target_include_directories(llama_shim PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${PREBUILT_ROOT}/include")

target_link_libraries(llama_shim PRIVATE ${LLAMA_LIBS} log)

# The shim's exported symbols are the only thing Dart looks up, and they are already marked
# with __attribute__((visibility("default"))) __attribute__((used)) in the header. Hiding
# everything else keeps the library small and prevents symbol collisions with the app.
set_target_properties(llama_shim PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON)

# Gradle packages targets built by CMake automatically, but IMPORTED targets are not built,
# so the prebuilt .so files have to be copied next to the shim explicitly.
foreach(LIB ${LLAMA_LIBS})
    add_custom_command(TARGET llama_shim POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${PREBUILT_ABI}/lib${LIB}.so"
            "$<TARGET_FILE_DIR:llama_shim>/lib${LIB}.so")
endforeach()
